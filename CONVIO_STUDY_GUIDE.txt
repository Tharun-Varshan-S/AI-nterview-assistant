export function useReducedMotion() {
  const [reduced, setReduced] = useState(false);

  useEffect(() => {
    const media = window.matchMedia('(prefers-reduced-motion: reduce)');
    setReduced(media.matches);
    const handler = () => setReduced(media.matches);
    media.addEventListener('change', handler);
    return () => media.removeEventListener('change', handler);
  }, []);

  return reduced;
}

WHAT IT DOES:
Checks if the user has "reduce motion" enabled in their OS settings.
Some users get dizzy from animations, so we respect their preference.

HOW IT WORKS:
1. window.matchMedia checks the user's system preference
2. If they want reduced motion, reduced = true
3. Other components can skip animations if reduced = true

EXAMPLE:
const reduced = useReducedMotion();
if (reduced) {
  // Show final state immediately
} else {
  // Play smooth animation
}

export function useCountUp(target: number, duration = 900, decimals = 0, start = true) {
  const reducedMotion = useReducedMotion();
  const [value, setValue] = useState(0);

  useEffect(() => {
    if (!start) {
      setValue(0);
      return;
    }

    if (reducedMotion) {
      setValue(target);
      return;
    }

    let rafId = 0;
    let startTime = 0;

    const animate = (timestamp: number) => {
      if (!startTime) {
        startTime = timestamp;
      }

      const progress = Math.min((timestamp - startTime) / duration, 1);
      const eased = 1 - Math.pow(1 - progress, 3);
      setValue(target * eased);

      if (progress < 1) {
        rafId = window.requestAnimationFrame(animate);
      }
    };

    rafId = window.requestAnimationFrame(animate);
    return () => window.cancelAnimationFrame(rafId);
  }, [target, duration, reducedMotion, start]);

  return useMemo(() => Number(value.toFixed(decimals)), [value, decimals]);
}

WHAT IT DOES:
Animates a number from 0 to target (e.g., 0 → 1234)

HOW IT WORKS STEP-BY-STEP:

1. START:
   - value starts at 0
   - target = 1234 (what we want to reach)
   - duration = 900ms (how long animation takes)

2. ANIMATION LOOP:
   - requestAnimationFrame calls animate() every frame (~60 times per second)
   - timestamp = current time in milliseconds
   - progress = how far through animation (0 to 1)
     Example: After 450ms, progress = 450/900 = 0.5 (halfway)

3. EASING:
   - eased = 1 - Math.pow(1 - progress, 3)
   - This creates "ease out" effect (fast start, slow end)
   - Example: progress=0.5 → eased=0.875 (moves faster at start)

4. UPDATE VALUE:
   - setValue(target * eased)
   - Example: 1234 * 0.875 = 1080
   - React re-renders with new value

5. REPEAT:
   - If progress < 1, call animate() again next frame
   - When progress = 1, animation stops

WHY requestAnimationFrame?
- Syncs with browser's refresh rate (smooth 60fps)
- Pauses when tab is hidden (saves battery)
- GPU-optimized

EXAMPLE USAGE:
const count = useCountUp(1234, 900, 0, true);
return <span>{count}</span>;  // Shows: 0 → 123 → 456 → 789 → 1234

export function useRafProgress(progress: number, duration = 700) {
  const reducedMotion = useReducedMotion();
  const [current, setCurrent] = useState(0);

  useEffect(() => {
    if (reducedMotion) {
      setCurrent(progress);
      return;
    }

    let rafId = 0;
    let startTime = 0;
    const startValue = current;
    const delta = progress - startValue;

    const animate = (timestamp: number) => {
      if (!startTime) {
        startTime = timestamp;
      }
      const t = Math.min((timestamp - startTime) / duration, 1);
      const eased = 1 - Math.pow(1 - t, 3);
      setCurrent(startValue + delta * eased);
      if (t < 1) {
        rafId = requestAnimationFrame(animate);
      }
    };

    rafId = requestAnimationFrame(animate);
    return () => cancelAnimationFrame(rafId);
  }, [progress, duration, reducedMotion]);

  return current;
}

WHAT IT DOES:
Smoothly animates progress bar from current value to new value

DIFFERENCE FROM useCountUp:
- useCountUp: Always starts from 0
- useRafProgress: Starts from current value

EXAMPLE:
Progress bar at 30%, user completes task, now 60%
- Without animation: 30% → 60% (instant jump)
- With useRafProgress: 30% → 35% → 40% → 45% → ... → 60% (smooth)

HOW IT WORKS:
1. startValue = 30 (current position)
2. progress = 60 (new target)
3. delta = 60 - 30 = 30 (how much to move)
4. Each frame: current = 30 + (30 * eased)
5. Result: Smooth transition from 30 to 60

COMPONENT 4: MicroButton (WITH RIPPLE EFFECT)
-------------------------------------------------------------------------------

import { ButtonHTMLAttributes, MouseEvent, useState } from 'react';

interface Ripple {
  x: number;
  y: number;
  id: number;
}

type MicroButtonProps = ButtonHTMLAttributes<HTMLButtonElement> & {
  glow?: boolean;
};

export default function MicroButton({ 
  children, 
  className = '', 
  glow = false, 
  onClick, 
  ...rest 
}: MicroButtonProps) {
  const [ripples, setRipples] = useState<Ripple[]>([]);

  const handleClick = (e: MouseEvent<HTMLButtonElement>) => {
    const rect = e.currentTarget.getBoundingClientRect();
    const ripple = { 
      x: e.clientX - rect.left, 
      y: e.clientY - rect.top, 
      id: Date.now() 
    };
    setRipples((prev) => [...prev, ripple]);
    window.setTimeout(() => {
      setRipples((prev) => prev.filter((r) => r.id !== ripple.id));
    }, 450);
    onClick?.(e);
  };

  return (
    <button
      {...rest}
      onClick={handleClick}
      className={`group relative overflow-hidden rounded-xl px-5 py-2.5 font-medium transition-all duration-300 ease-[cubic-bezier(0.2,0.8,0.2,1)] hover:-translate-y-0.5 hover:shadow-convio-card-hover disabled:translate-y-0 disabled:opacity-60 ${
        glow ? 'shadow-convio-glow' : ''
      } ${className}`}
    >
      <span className="relative z-10 inline-flex items-center gap-2">{children}</span>
      {ripples.map((ripple) => (
        <span
          key={ripple.id}
          className="pointer-events-none absolute h-20 w-20 -translate-x-1/2 -translate-y-1/2 rounded-full bg-white/30 animate-ripple"
          style={{ left: ripple.x, top: ripple.y }}
        />
      ))}
    </button>
  );
}

WHAT IT DOES:
Button with Material Design ripple effect on click

HOW RIPPLE WORKS STEP-BY-STEP:

1. USER CLICKS BUTTON:
   - Click happens at screen position (500px, 300px)

2. CALCULATE CLICK POSITION RELATIVE TO BUTTON:
   const rect = e.currentTarget.getBoundingClientRect();
   // rect = { left: 400, top: 250, width: 100, height: 50 }
   
   x = e.clientX - rect.left = 500 - 400 = 100px (from left edge)
   y = e.clientY - rect.top = 300 - 250 = 50px (from top edge)

3. CREATE RIPPLE OBJECT:
   ripple = { x: 100, y: 50, id: 1234567890 }
   
4. ADD TO STATE:
   setRipples([...prev, ripple])
   // This triggers re-render

5. RENDER RIPPLE:
   <span style={{ left: 100, top: 50 }} />
   // Circle appears at click position
   // CSS animation expands it outward

6. REMOVE AFTER 450ms:
   setTimeout(() => {
     setRipples(prev => prev.filter(r => r.id !== 1234567890))
   }, 450)

CSS ANIMATION (in tailwind.config.ts):
ripple: {
  '0%': { transform: 'scale(0.25)', opacity: '0.45' },
  '100%': { transform: 'scale(1.6)', opacity: '0' }
}

RESULT:
- Circle starts small and visible
- Expands to 1.6x size
- Fades out
- Gets removed from DOM

USAGE:
<MicroButton glow onClick={() => alert('Clicked!')}>
  Submit
</MicroButton>

import { Children, ReactNode } from 'react';

interface StaggerContainerProps {
  children: ReactNode;
  className?: string;
  delayStepMs?: number;
}

export default function StaggerContainer({ 
  children, 
  className = '', 
  delayStepMs = 80 
}: StaggerContainerProps) {
  const list = Children.toArray(children);

  return (
    <div className={className}>
      {list.map((child, index) => (
        <div
          key={index}
          className="animate-fade-up"
          style={{ 
            animationDelay: `${index * delayStepMs}ms`, 
            animationFillMode: 'both' 
          }}
        >
          {child}
        </div>
      ))}
    </div>
  );
}

WHAT IT DOES:
Makes list items appear one after another (cascade effect)

HOW IT WORKS:

1. CONVERT CHILDREN TO ARRAY:
   const list = Children.toArray(children);
   // [<Card>1</Card>, <Card>2</Card>, <Card>3</Card>]

2. WRAP EACH CHILD:
   Each child gets wrapped in a div with animation

3. CALCULATE DELAY:
   - Item 0: delay = 0 * 80 = 0ms
   - Item 1: delay = 1 * 80 = 80ms
   - Item 2: delay = 2 * 80 = 160ms

4. ANIMATION:
   animate-fade-up: Fades in + slides up
   animationFillMode: 'both' keeps initial state (invisible) until animation starts

RESULT:
Item 1 appears at 0ms
Item 2 appears at 80ms
Item 3 appears at 160ms
Creates smooth cascade effect

USAGE:
<StaggerContainer delayStepMs={100}>
  <Card>Interview 1</Card>
  <Card>Interview 2</Card>
  <Card>Interview 3</Card>
</StaggerContainer>

}

WHAT IT DOES:
Animated underline that slides in on hover

HOW IT WORKS:

1. INITIAL STATE:
   - ::after creates invisible line below text
   - transform: scaleX(0): Width = 0 (invisible)
   - transform-origin: left: Grows from left

2. ON HOVER:
   - transform: scaleX(1): Width = 100%
   - Transition: Smooth 320ms animation

3. RESULT:
   Text → Text (underline slides in from left)

USAGE:
<a className="convio-link-line">Hover me</a>

PART 7: KEY CONCEPTS TO REMEMBER
===============================================================================

1. requestAnimationFrame:
   - Browser API for smooth animations
   - Runs at 60fps
   - GPU-optimized

2. Easing Functions:
   - Linear: Constant speed
   - Ease-out: Fast start, slow end
   - cubic-bezier(0.2, 0.8, 0.2, 1): Custom smooth curve

3. Transform vs Position:
   - transform: translateY(-4px) → GPU-accelerated, smooth
   - top: -4px → CPU-based, can be janky
   - ALWAYS use transform for animations

4. Opacity vs Visibility:
   - opacity: 0 → Can animate smoothly
   - visibility: hidden → Cannot animate
   - Use opacity for fade effects

5. Glassmorphism:
   - Semi-transparent background
   - backdrop-filter: blur()
   - Subtle border and shadow

6. Stagger Animation:
   - Delay each item by fixed amount
   - Creates cascade effect
   - Makes lists feel alive

7. Ripple Effect:
   - Track click position
   - Create expanding circle
   - Remove after animation

8. Count-up Animation:
   - Start at 0
   - Increment each frame
   - Use easing for smooth motion
